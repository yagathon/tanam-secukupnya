> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A second page, `/sandbox`, where the founder sets up a blank farm (crop growth curve, lines, buyers with usual demand, optional existing beds), then logs real events day by day (form or Gemini note → confirm card → OK) and watches the engine's decisions move, with a per-day decision trail. The engine also switches to growth model B (per-bed size factor) and gains slow reference-curve learning.

**Architecture:** The sandbox is the same engine, replayed from a different reference: `SandboxStore` wraps the visitor's store, derives `crops/buyers/households/batches/history/meta` from a saved setup doc, and keeps the log in `sandbox_events`/`sandbox_notes`. Every existing view handler is re-registered under `/api/sandbox/*` with that store, so Today/Beds/Tomorrow/Sow/Demand come for free. Day close in the sandbox uses `sandbox.close_day` (user-entered orders and harvests, zero-filled actuals) instead of the simulated world.

**Tech Stack:** FastAPI + pydantic v2, pure-Python engine (pytest via `uv`), Next.js App Router + TypeScript (vitest, eslint, recharts 3).

**Spec:** `docs/superpowers/specs/2026-10-15-sandbox-design.md`

## Global Constraints

- Everything visible in the hackathon submission is English (UI copy, code, docs).
- Seed and eval data use aliases only (Restaurant A, Bed 3, …) — never real names or phone numbers.
- The existing demo (`/`) keeps working and keeps its behaviour, except the model-B switch (Task 1), which is applied to both on purpose; reference learning is OFF in the demo.
- Model B: `avg_g = size_factor × reference(bed_day)`; `size_factor` clamped to `[0.5, 1.5]`.
- Reference learning: healthy beds only; per node keep the last 5 ratios; `w1 = (t − x0)/(x1 − x0)`; `y ← y × (1 + clamp(0.3·(median−1)·w, ±5%))`; monotone repair `y[i] ≥ y[i−1] + 1 g`; the sampled bed is then re-anchored against the new reference.
- Cold-start demand: level = usual per week ÷ number of order weekdays (all 7 if none given); σ = 50% of level until 5 residuals exist; no estimate → level 0 until the first order; market factor needs ≥ 3 businesses (already `MARKET_MIN_BUYERS = 3`).
- Limits: ≤ 400 days, ≤ 50 buyers, ≤ 50 beds at start, growth curve 2–30 points; validation errors are HTTP 422.
- Nothing from a Gemini note is stored as an event before OK.
- Confirm-card formats (exact): `ORDER · Restaurant A · regular · 12.0 kg · Fri 16 Oct` and `HARVEST · Bed 3 · 20.0 kg · sample 640 g / 410 g`.
- Currency is not shown anywhere in the sandbox (kg only).
- Merge/deploy only with the founder's go-ahead. Never commit `frontend/public/probe.html`. Never run `next build` while `next dev` runs.

```python
"""Slow learning of the reference growth curve from harvest samples (model B, layer 2).

A bed's own size factor absorbs its stunting; the reference only moves when many healthy beds at
the same age disagree with it. Each node keeps the last few ratios (sample avg ÷ reference) it was
near, and moves by a capped fraction of their median — so one odd bed cannot bend the curve.
"""

from __future__ import annotations

import statistics

Pts = tuple[tuple[float, float], ...]
Kept = tuple[tuple[float, ...], ...]


def node_weights(xs: list[float], t: float) -> list[float]:
    """Linear weights of the two nodes around bed day t; outside the curve the end node gets 1."""
    w = [0.0] * len(xs)
    if t <= xs[0]:
        w[0] = 1.0
    elif t >= xs[-1]:
        w[-1] = 1.0
    else:
        i = next(i for i in range(len(xs) - 1) if xs[i] <= t <= xs[i + 1])
        w1 = (t - xs[i]) / (xs[i + 1] - xs[i])
        w[i], w[i + 1] = 1 - w1, w1
    return w


def update(pts: Pts, kept: Kept, t: float, ratio: float, keep: int = 5, gain: float = 0.3,
           cap: float = 0.05) -> tuple[Pts, Kept]:
    kept = kept or tuple(() for _ in pts)
    ws = node_weights([x for x, _ in pts], t)
    new_pts, new_kept = [], []
    for (x, y), k, w in zip(pts, kept, ws):
        if w > 0:
            k = (k + (ratio,))[-keep:]
            step = max(-cap, min(cap, gain * (statistics.median(k) - 1) * w))
            y = y * (1 + step)
        new_pts.append((x, y))
        new_kept.append(k)
    for i in range(1, len(new_pts)):                       # strictly increasing, by ≥ 1 g
        x, y = new_pts[i]
        if y < new_pts[i - 1][1] + 1:
            new_pts[i] = (x, new_pts[i - 1][1] + 1)
    return tuple(new_pts), tuple(new_kept)
```

```python
def today_wib() -> date:
    return datetime.now(WIB).date()


class LineIn(BaseModel):
    id: str = Field(pattern=r"^[a-z0-9_]{1,20}$")
    min_g: float = Field(gt=0)
    max_g: float = Field(gt=0)

    @model_validator(mode="after")
    def _order(self):
        if self.min_g >= self.max_g:
            raise ValueError(f"line {self.id}: min_g must be below max_g")
        return self


class CropIn(BaseModel):
    name: str = Field(min_length=1, max_length=40)
    nursery_days: int = Field(ge=0, le=60)
    growth_curve: list[tuple[float, float]] = Field(min_length=2, max_length=30)
    spread_ratio: float = Field(gt=0, le=2)
    germination: float = Field(gt=0, le=1)
    survival: float = Field(gt=0, le=1)
    seeds_per_gram: float = Field(gt=0)
    grams_per_teaspoon: float = Field(gt=0)
    max_bed_day: int = Field(ge=1, le=200)
    max_seedlings_per_sowing: int = Field(ge=1, le=100000)
    sow_weekdays: list[int] = Field(min_length=1, max_length=7)
    lines: list[LineIn] = Field(min_length=1, max_length=5)

    @field_validator("growth_curve")
    @classmethod
    def _increasing(cls, v):
        if any(b[0] <= a[0] or b[1] <= a[1] for a, b in zip(v, v[1:])):
            raise ValueError("growth curve: days and grams must both increase")
        if v[0][1] <= 0:
            raise ValueError("growth curve: grams must be above 0")
        return v

    @field_validator("sow_weekdays")
    @classmethod
    def _weekdays(cls, v):
        if any(not 0 <= d <= 6 for d in v):
            raise ValueError("sow weekdays are 0 (Mon) … 6 (Sun)")
        return sorted(set(v))


class BuyerIn(BaseModel):
    id: str | None = Field(None, pattern=r"^[a-z0-9_]{1,20}$")
    name: str = Field(min_length=1, max_length=40)
    type: Literal["business", "household"]
    line: str
    usual_kg_per_week: float = Field(0, ge=0, le=5000)
    weekdays: list[int] = Field(default_factory=list, max_length=7)
    min_kg: float = Field(0, ge=0, le=500)


class BedIn(BaseModel):
    line: str
    sown_on: date
    seedlings: int = Field(gt=0, le=100000)


class SetupIn(BaseModel):
    start: date = Field(default_factory=today_wib)
    learn_reference: bool = True
    crop: CropIn
    buyers: list[BuyerIn] = Field(default_factory=list, max_length=MAX_BUYERS)
    beds: list[BedIn] = Field(default_factory=list, max_length=MAX_BEDS)

    @model_validator(mode="after")
    def _consistent(self):
        lines = {ln.id for ln in self.crop.lines}
        names, ids = set(), set()
        for i, b in enumerate(self.buyers, 1):
            if b.line not in lines:
                raise ValueError(f"buyer {b.name}: unknown line {b.line!r}")
            if b.name in names:
                raise ValueError(f"buyer name {b.name!r} is used twice")
            names.add(b.name)
            if b.id is None:
                b.id = next(f"b{n}" for n in range(i, i + MAX_BUYERS + 1) if f"b{n}" not in ids
                            and all(o.id != f"b{n}" for o in self.buyers))
            if b.id in ids:
                raise ValueError(f"buyer id {b.id!r} is used twice")
            ids.add(b.id)
            if any(not 0 <= d <= 6 for d in b.weekdays):
                raise ValueError(f"buyer {b.name}: weekdays are 0 (Mon) … 6 (Sun)")
        for b in self.beds:
            if b.line not in lines:
                raise ValueError(f"bed: unknown line {b.line!r}")
            if b.sown_on > self.start:
                raise ValueError(f"bed sown {b.sown_on} is after the start {self.start}")
        return self

def _week(b: dict) -> list[float]:
    days = b["weekdays"] or list(range(7))
    per = b["usual_kg_per_week"] / len(days)
    return [round(per, 4) if d in days else 0.0 for d in range(7)]


def priors_from_setup(s: dict) -> dict[str, list[float]]:
    out: dict[str, list[float]] = {}
    for b in s["buyers"]:
        key = engine.b2b_key(b["id"], b["line"]) if b["type"] == "business" else engine.hh_key(b["line"])
        out[key] = [round(a + w, 4) for a, w in zip(out.get(key, [0.0] * 7), _week(b))]
    for ln in s["crop"]["lines"]:                 # a line with households but no estimate still gets a series
        if any(b["type"] == "household" and b["line"] == ln["id"] for b in s["buyers"]):
            out.setdefault(engine.hh_key(ln["id"]), [0.0] * 7)
    return out


def _key(s: dict) -> str:
    return hashlib.sha256(json.dumps(s, sort_keys=True).encode()).hexdigest()[:16]


@lru_cache(maxsize=64)
def _reference(raw: str) -> dict:
    s = json.loads(raw)
    start = date.fromisoformat(s["start"])
    crop = {"id": "crop", **{k: v for k, v in s["crop"].items()},
            "lines": [{**ln, "channels": ["B2B", "B2C"]} for ln in s["crop"]["lines"]]}
    biz = [b for b in s["buyers"] if b["type"] == "business"]
    hh = [b for b in s["buyers"] if b["type"] == "household"]
    beds = sorted(s["beds"], key=lambda b: b["sown_on"])
    batches = [{"id": f"{b['line']}-{b['sown_on']}-{i}", "label": f"Bed {i}", "line": b["line"],
                "sown_on": b["sown_on"], "seedlings": b["seedlings"]} for i, b in enumerate(beds, 1)]
    last = {ln["id"]: max((b["sown_on"] for b in beds if b["line"] == ln["id"]), default=None) for ln in s["crop"]["lines"]}
    return {
        "crops": [crop],
        "buyers": [{"id": b["id"], "alias": b["name"],
                    "lines": {b["line"]: {"kg": round(sum(_week(b)) / max(1, sum(1 for v in _week(b) if v)), 2),
                                          "min_kg": b["min_kg"]}}} for b in biz],
        "households": [{"id": b["id"], "alias": b["name"], "line": b["line"],
                        "kg": round(b["usual_kg_per_week"] / 7, 2), "weekdays": b["weekdays"]} for b in hh],
        "batches": batches,
        "history": [],
        "meta": {"day0": start.isoformat(), "shape_until": start.isoformat(), "sandbox": True, "synthetic": False,
                 "learn_reference": s["learn_reference"], "priors": priors_from_setup(s),
                 "household_ids": [b["id"] for b in hh], "rev": _key(s),
                 "last_sown": {k: v for k, v in last.items() if v}, "next_bed": len(beds) + 1},
    }


def reference(setup: dict) -> dict:
    return _reference(json.dumps(setup, sort_keys=True))

class SandboxStore:
    """The visitor's store, seen through the sandbox: reference from the setup, log in sandbox_*."""

    def __init__(self, base: Store, setup: dict | None):
        self.base, self.setup = base, setup or SetupIn.model_validate(default_setup()).model_dump(mode="json")
        self.saved = setup is not None
        self.uid = f"{base.uid}#sandbox"

    def list(self, collection: str) -> list[dict]:
        if collection in DERIVED:
            return reference(self.setup)[collection]
        return self.base.list(f"sandbox_{collection}")

    def get(self, collection: str, id: str) -> dict | None:
        return self.base.get(f"sandbox_{collection}", id)

    def put(self, collection: str, doc: dict) -> dict:
        return self.base.put(f"sandbox_{collection}", doc)

    def add(self, collection: str, doc: dict) -> bool:
        return self.base.add(f"sandbox_{collection}", doc)

    def meta(self) -> dict:
        return reference(self.setup)["meta"]

    def reset(self, collections: tuple[str, ...] = SANDBOX) -> None:
        self.base.reset(collections)

    def for_user(self, uid: str) -> "SandboxStore":
        return SandboxStore(self.base.for_user(uid), self.setup)


def open_sandbox(base: Store) -> SandboxStore:
    doc = base.get("sandbox_setup", SETUP_ID)
    return SandboxStore(base, doc["setup"] if doc else None)
```

```python
def close_day(st, crop, buyers: list[dict], households: list[dict], plans) -> list[dict]:
    """Today (t) → tomorrow (t+1) from what was entered: tomorrow's orders are served from the harvest
    logged for it (or, if none, the engine's planned send, harvested oldest-first); every buyer's
    actual is their order, or 0 on one of their regular weekdays."""
    t = st.today
    d = t + timedelta(days=1)
    iso = d.isoformat()
    by_id = {b["id"]: b for b in buyers}
    hh_ids = {h["id"] for h in households}
    evs: list[dict] = []
    for ln in crop.lines:
        line = ln.id
        orders = {(who, kg) for (day, who, l2), kg in st.known_orders.items() if day == iso and l2 == line}
        b2b = [BuyerOrder(w, kg, by_id[w]["lines"][line]["min_kg"]) for w, kg in sorted(orders) if w in by_id]
        hh = [HouseholdOrder(w, kg, st.last_shorted.get(w)) for w, kg in sorted(orders) if w in hh_ids]
        logged = st.harvested.get((t.isoformat(), line), 0.0) + st.harvested.get((iso, line), 0.0)
        got = logged
        if not logged and plans[line].send > 0:
            shadows = [replace(b) for b in st.alive(crop, line)]
            for b, kg in take_oldest(crop, shadows, d, plans[line].send):
                evs.append({"t": "harvest", "batch": b.id, "day": iso, "kg": round(kg, 2)})
                got += round(kg, 2)
        rows_b, rows_h = allocate_day(got, b2b, hh, d)
        evs += [{"t": "short", "household": r.customer_id, "day": iso} for r in rows_h if r.short_kg > 0]
        evs.append({"t": "served", "day": iso, "line": line,
                    "b2b_ordered": round(sum(r.ordered_kg for r in rows_b), 2),
                    "b2b_served": round(sum(r.served_kg for r in rows_b), 2),
                    "hh_ordered": round(sum(r.ordered_kg for r in rows_h), 2),
                    "hh_served": round(sum(r.served_kg for r in rows_h), 2), "waste_kg": 0.0})
        for b in buyers:
            if line not in b["lines"]:
                continue
            kg = dict(orders).get(b["id"])
            regular = d.weekday() in (b.get("weekdays") or range(7))
            if kg is not None or regular:
                evs.append({"t": "actual", "series": engine.b2b_key(b["id"], line), "day": iso, "kg": kg or 0.0})
        if any(h["line"] == line for h in households):
            evs.append({"t": "actual", "series": engine.hh_key(line), "day": iso,
                        "kg": round(sum(kg for w, kg in orders if w in hh_ids), 2)})
    evs.append({"t": "clock", "day": iso})
    return evs
```

```python
def describe(ev: dict, names: dict[str, str], labels: dict[str, str]) -> str:
    """One fixed-format line per event, for the confirm card and the trail."""
    who = lambda i: names.get(i, f"⚠ unknown buyer {i}")
    bed = lambda i: labels.get(i, f"⚠ unknown bed {i}")
    t = ev["t"]
    if t == "order":
        return f"ORDER · {who(ev['buyer'])} · {ev['line']} · {ev['kg']:.1f} kg · {_day(ev['day'])}"
    if t == "harvest":
        s = f" · sample {ev['big5_g']:.0f} g / {ev['small5_g']:.0f} g" if ev.get("big5_g") else ""
        return f"HARVEST · {bed(ev['batch'])} · {ev['kg']:.1f} kg{s}"
    if t == "sow":
        return f"SOW · {ev['line']} · {ev['seedlings']} seedlings · {_day(ev['day'])}"
    if t == "postpone":
        return f"POSTPONE · {ev['line']} · sowing of {_day(ev['day'])} · {ev.get('reason', '')}".rstrip(" ·")
    if t == "loss":
        return f"LOSS · {bed(ev['batch'])} · {ev['pct']:.0f}% · {ev.get('cause', '')}".rstrip(" ·")
    if t == "signal":
        return (f"SIGNAL · {ev['scope']}{' · ' + ev['line'] if ev.get('line') else ''} · {ev['effect_pct']:+.0f}% · "
                f"{_day(ev['date_from'])}–{_day(ev['date_to'])}")
    if t == "reference_reset":
        return "RESET · reference curve back to the starting curve"
    return t.upper()
```

```python
def form_events(raw: dict, st, crop, labels: dict[str, str], buyer_ids: set[str]) -> list[dict]:
    """One form submission → primitive events dated the sandbox's today. Raises ValueError with a
    sentence for people."""
    k, t = raw.get("kind"), st.today
    lines = {ln.id for ln in crop.lines}
    def line_of(x):
        if x not in lines:
            raise ValueError(f"unknown line {x!r}")
        return x
    def kg_of(x, hi=500.0):
        x = float(x)
        if not 0 < x <= hi:
            raise ValueError(f"kg must be above 0 and at most {hi:g}")
        return round(x, 2)
    def bed_of(x):
        if x not in st.batches:
            raise ValueError(f"unknown bed {x!r}")
        return x
    if k == "order":
        d = date.fromisoformat(raw["day"])
        if not t < d <= t + timedelta(days=14):
            raise ValueError("an order is for tomorrow or up to 14 days ahead")
        if raw.get("buyer") not in buyer_ids:
            raise ValueError(f"unknown buyer {raw.get('buyer')!r}")
        return [{"t": "order", "day": d.isoformat(), "buyer": raw["buyer"], "line": line_of(raw["line"]), "kg": kg_of(raw["kg"])}]
    if k == "harvest":
        ev = {"t": "harvest", "batch": bed_of(raw.get("batch")), "day": t.isoformat(), "kg": kg_of(raw["kg"], 5000)}
        if raw.get("big5_g") or raw.get("small5_g"):
            big, small = float(raw["big5_g"]), float(raw["small5_g"])
            if not big >= small > 0:
                raise ValueError("sample: the 5 biggest must weigh at least the 5 smallest, both above 0")
            ev |= {"big5_g": big, "small5_g": small}
        return [ev]
    if k == "sow":
        n = int(raw.get("seedlings", 0))
        if not 0 < n <= crop.max_seedlings:
            raise ValueError(f"seedlings must be 1 … {crop.max_seedlings}")
        return [{"t": "sow", "line": line_of(raw["line"]), "day": t.isoformat(), "seedlings": n}]
    if k == "postpone":
        line = line_of(raw["line"])
        sown = st.last_sown.get(line) == t or (line, t.isoformat()) in st.postponed
        day = engine.next_sow_day(t, sown, crop.sow_weekdays)
        return [{"t": "postpone", "line": line, "day": day.isoformat(), "reason": str(raw.get("reason", ""))[:80]}]
    if k == "loss":
        pct = float(raw["pct"])
        if not 0 < pct <= 100:
            raise ValueError("loss % must be above 0 and at most 100")
        cause = str(raw.get("cause", ""))[:80]
        ids = [bed_of(raw["batch"])] if raw.get("batch") else [
            b.id for b in st.alive(crop, line_of(raw["line"])) if bed_day(crop, b, t) >= 0 and in_bed_plants(crop, b, t) > 0]
        if not ids:
            raise ValueError("no bed in the ground on that line")
        return [{"t": "loss", "batch": i, "day": t.isoformat(), "pct": pct, "cause": cause} for i in ids]
    if k == "signal":
        scopes = {"all", "channel:B2B", "channel:B2C"} | {f"buyer:{b}" for b in buyer_ids}
        if raw.get("scope") not in scopes:
            raise ValueError(f"unknown scope {raw.get('scope')!r}")
        line = line_of(raw["line"]) if raw.get("line") else None
        pct = float(raw["effect_pct"])
        if not -100 <= pct <= 300:
            raise ValueError("effect % must be between -100 and +300")
        d0 = date.fromisoformat(raw["date_from"])
        d1 = date.fromisoformat(raw["date_to"]) if raw.get("date_to") else t + timedelta(days=182)
        if not t <= d0 <= d1 <= t + timedelta(days=182):
            raise ValueError("signal dates run from today, up to 182 days ahead, and from ≤ to")
        return [{"t": "signal", "scope": raw["scope"], "line": line, "effect_pct": round(pct, 1),
                 "date_from": d0.isoformat(), "date_to": d1.isoformat(), "cause": str(raw.get("cause", ""))[:120]}]
    if k == "reference_reset":
        return [{"t": "reference_reset", "day": t.isoformat()}]
    raise ValueError(f"unknown form {k!r}")
```

```python
"""The decision trail: for every closed day, what the engine decided that morning and what came in.

Replays the sandbox log once, planning at each day boundary; cached per workspace and continued
from the last day when the log only grew (a setup edit or a clear rebuilds it)."""

from __future__ import annotations

import copy
from dataclasses import dataclass, field

from app import farm, sandbox, sim
from app.state import apply, current_crop, initial
from app.store import Store


@dataclass
class _Cache:
    rev: str
    seqs: list[int]
    state: object
    rows: list[dict] = field(default_factory=list)
    pending: list[str] = field(default_factory=list)
    ref_seen: int = 0


_cache: dict[str, _Cache] = {}


def _row(st, crop, buyers, inputs, ref, live) -> dict:
    plans = sim.plan(st, crop, buyers)
    lines = {}
    for line, p in plans.items():
        r0 = p.needs.rows[0] if p.needs.rows else {"low": 0, "high": 0}
        lines[line] = {"low": r0["low"], "high": r0["high"], "ready": round(p.available_tomorrow, 2),
                       "ceiling": round(p.ceiling, 2), "send": round(p.send, 2), "short": None,
                       "sow_day": p.sow.day.isoformat() if p.sow.day else None, "sow_seedlings": p.sow.seedlings}
    return {"day": st.today.isoformat(), "live": live, "lines": lines, "inputs": inputs, "ref": ref}


def build(ws: Store) -> dict:
    meta, docs = ws.meta(), sorted(ws.list("events"), key=lambda d: d["seq"])
    c = farm.crop(ws)
    buyers = ws.list("buyers")
    names = {b["id"]: b["alias"] for b in buyers} | {h["id"]: h["alias"] for h in ws.list("households")}
    hit = _cache.get(ws.uid)
    seqs = [d["seq"] for d in docs]

        st = initial({"batches": ws.list("batches"), "history": ws.list("history")}, meta)
        hit = _Cache(meta["rev"], [], st)
    start = len(hit.seqs)
    st = hit.state
    for doc in docs[start:]:
        labels = {b.id: b.label for b in st.batches.values()}
        if doc["kind"] == "day":
            row = _row(st, current_crop(st, c), buyers, hit.pending,
                       [_ref_line(e) for e in st.ref_log[hit.ref_seen:]], False)
            hit.ref_seen, hit.pending = len(st.ref_log), []
        for ev in doc["events"]:
            if doc["kind"] != "day":
                hit.pending.append(sandbox.describe(ev, names, labels))
            if doc["kind"] == "day" and ev["t"] == "served":
                row["lines"][ev["line"]]["short"] = round(ev["b2b_ordered"] + ev["hh_ordered"]
                                                          - ev["b2b_served"] - ev["hh_served"], 2)
            apply(st, c, ev)
        if doc["kind"] == "day":
            hit.rows.append(row)
        hit.seqs.append(doc["seq"])
    _cache[ws.uid] = hit
    live = _row(st, current_crop(st, c), buyers, list(hit.pending),
                [_ref_line(e) for e in st.ref_log[hit.ref_seen:]], True)
    return {"rows": [live] + hit.rows[::-1], "replayed_from": start}


def _ref_line(e: dict) -> str:

        return "reference reset to the starting curve"
    return f"{e['bed']} sample ×{e['ratio']:.2f} → nodes at day {', '.join(str(int(x)) for x in e['node_days'])}" \
        if e["node_days"] else f"{e['bed']} sample ×{e['ratio']:.2f} (reference held)"
```

```ts
export function makeApi(base: "" | "/sandbox") {
  const p = (s: string) => `${base}${s}`;
  return {
    state: () => call<State>(p("/state")),
    today: () => call<Today>(p("/today")),
    beds: () => call<Beds>(p("/beds")),
    tomorrow: () => call<Tomorrow>(p("/tomorrow")),
    sow: () => call<Sow>(p("/sow")),
    demand: () => call<Demand>(p("/demand")),
    season: () => call<Season>("/season"),
    readNote: (text: string, sender: string) => call<NoteDoc>(p("/notes"), { text, sender }),
    why: () => call<{ text: string; model: string; cached: boolean }>(p("/today/why"), {}),
    decideNote: (id: string, accept: boolean) => call<NoteDoc>(p(`/notes/${id}/${accept ? "accept" : "reject"}`), {}),
    nextDay: (from_day: string, days = 1) => call<{ today: string; appended: number }>(p("/day/next"), { from_day, days }),
    reset: () => call<{ ok: boolean }>(base ? "/sandbox/clear" : "/demo/reset", {}),
    setup: () => call<SetupDoc>("/sandbox/setup"),
    saveSetup: (s: Setup) => call<SetupDoc>("/sandbox/setup", s, "PUT"),
    event: (form: Record<string, unknown>, seq: number) => call<{ events: unknown[]; lines: string[] }>("/sandbox/events", { form, seq }),
    trail: () => call<{ rows: TrailRow[] }>("/sandbox/trail"),
  };
}
export type Api = ReturnType<typeof makeApi>;
export const api = makeApi("");
```

```ts
import type { SetupBed, SetupBuyer } from "./types";

export const WEEKDAYS = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"];

export function parseSheet(text: string): string[][] {
  const lines = text.replace(/^﻿/, "").split(/\r?\n/).filter((l) => l.trim() !== "");
  const sep = lines.some((l) => l.includes("\t")) ? "\t" : lines.some((l) => l.includes(";")) ? ";" : ",";
  return lines.map((l) => l.split(sep).map((c) => c.trim()));
}

export function toNumber(s: string): number | null {
  let t = s.trim().replace(/\s/g, "");
  if (t === "") return null;
  if (t.includes(",") && t.includes(".")) t = t.replace(/\./g, "").replace(",", ".");
  else if (t.includes(",")) t = t.replace(",", ".");
  const n = Number(t);
  return Number.isFinite(n) ? n : null;
}

export function toSheet(rows: (string | number)[][]): string {
  return rows.map((r) => r.join("\t")).join("\n");
}

export function parseWeekdays(s: string): number[] {
  const out = new Set<number>();
  for (const w of s.split(/[\s,;/]+/).filter(Boolean)) {
    const i = WEEKDAYS.findIndex((d) => d.toLowerCase() === w.slice(0, 3).toLowerCase());
    if (i >= 0) out.add(i);
  }
  return [...out].sort((a, b) => a - b);
}

function body(rows: string[][]): string[][] {
  return rows.length && isHeader(rows[0]) ? rows.slice(1) : rows;
}

const HEADERS = ["day", "name", "line", "grams", "type"];
function isHeader(r: string[]): boolean {
  return r.some((c) => HEADERS.includes(c.toLowerCase().split(/\s/)[0]));
}

export function curveFromSheet(text: string): { points: [number, number][] } | { error: string } {
  const rows = body(parseSheet(text));
  const pts: [number, number][] = [];
  for (const [i, r] of rows.entries()) {
    const d = toNumber(r[0] ?? ""), g = toNumber(r[1] ?? "");
    if (d === null) return { error: `row ${i + 1}: day is not a number` };
    if (g === null) return { error: `row ${i + 1}: grams is not a number` };
    if (pts.length && (d <= pts[pts.length - 1][0] || g <= pts[pts.length - 1][1]))
      return { error: `row ${i + 1}: days and grams must both increase` };
    pts.push([d, g]);
  }
  return { points: pts };
}

export function buyersFromSheet(text: string, lines: string[]): { buyers: SetupBuyer[] } | { error: string } {
  const out: SetupBuyer[] = [];
  for (const [i, r] of body(parseSheet(text)).entries()) {
    const [name, type, line, usual = "", wd = "", min = ""] = r;
    if (!name) return { error: `row ${i + 1}: name is empty` };
    if (type !== "business" && type !== "household") return { error: `row ${i + 1}: type must be business or household` };
    if (!lines.includes(line)) return { error: `row ${i + 1}: unknown line ${line}` };
    const u = usual === "" ? 0 : toNumber(usual), m = min === "" ? 0 : toNumber(min);
    if (u === null || u < 0) return { error: `row ${i + 1}: usual kg/week is not a number` };
    if (m === null || m < 0) return { error: `row ${i + 1}: min kg is not a number` };
    out.push({ name, type, line, usual_kg_per_week: u, weekdays: parseWeekdays(wd), min_kg: m });
  }
  return { buyers: out };
}

export function bedsFromSheet(text: string, lines: string[]): { beds: SetupBed[] } | { error: string } {
  const out: SetupBed[] = [];
  for (const [i, r] of body(parseSheet(text)).entries()) {
    const [line, sown, n] = r;
    if (!lines.includes(line)) return { error: `row ${i + 1}: unknown line ${line}` };
    if (!/^\d{4}-\d{2}-\d{2}$/.test(sown ?? "")) return { error: `row ${i + 1}: sown on must be YYYY-MM-DD` };
    const s = toNumber(n ?? "");
    if (s === null || s <= 0 || !Number.isInteger(s)) return { error: `row ${i + 1}: seedlings must be a whole number above 0` };
    out.push({ line, sown_on: sown, seedlings: s });
  }
  return { beds: out };
}
```

```ts
import type { Setup } from "./types";

/** The server's setup rules, client-side, so problems show while typing (the server stays the judge). */
export function checkSetup(s: Setup): string[] {
  const p: string[] = [];
  const c = s.crop;
  if (c.growth_curve.length < 2 || c.growth_curve.length > 30) p.push("growth curve needs 2 to 30 points");
  c.growth_curve.forEach(([d, g], i) => {
    if (i && (d <= c.growth_curve[i - 1][0] || g <= c.growth_curve[i - 1][1])) p.push(`curve point ${i + 1}: days and grams must both increase`);
  });
  if (!c.sow_weekdays.length) p.push("pick at least one sowing weekday");
  const lines = new Set(c.lines.map((l) => l.id));
  for (const l of c.lines) if (l.min_g >= l.max_g) p.push(`line ${l.id}: min g must be below max g`);
  const names = new Set<string>();
  if (s.buyers.length > 50) p.push("at most 50 buyers");
  for (const b of s.buyers) {
    if (names.has(b.name)) p.push(`buyer name ${b.name} is used twice`);
    names.add(b.name);
    if (!lines.has(b.line)) p.push(`buyer ${b.name}: unknown line ${b.line}`);
  }
  for (const b of s.beds) {
    if (!lines.has(b.line)) p.push(`bed: unknown line ${b.line}`);
    if (b.sown_on > s.start) p.push(`bed sown ${b.sown_on} is after the start ${s.start}`);
  }
  return p;
}
```
